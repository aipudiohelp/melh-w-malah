       <div id="cateringNoticeBox" class="hidden bg-amber-950/60 border-2 border-amber-500/80 p-3.5 rounded-2xl space-y-2">
         <div class="flex items-start gap-2">

 <script>
   const WHATSAPP_PHONE = "201145295914";
   const DELIVERY_STANDARD_FEE = 75;

   function scrollSlider(direction) {
     const slider = document.getElementById('mealsSlider');
     const scrollAmount = 280;
     slider.scrollBy({
       left: direction * -scrollAmount,
       behavior: 'smooth'
     });
   }

   let currentOrder = {
     title: "",
     unitPrice: 0,
     quantity: 1,
     isCatering: false,
     total: 0
   };

   function openOrderModal(title, price = 0, isCatering = false) {
     currentOrder.title = title;
     currentOrder.unitPrice = price;
     currentOrder.quantity = 1;
     currentOrder.isCatering = isCatering || title.includes("عزوم") || title.includes("العزومات");

     document.getElementById('orderQtyDisplay').textContent = currentOrder.quantity;
     document.querySelectorAll('.addon-checkbox').forEach(cb => cb.checked = false);

     document.getElementById('modalItemTitle').textContent = currentOrder.title;

     const cateringBox = document.getElementById('cateringNoticeBox');
     const depositCheck = document.getElementById('depositAgreement');

     if (currentOrder.isCatering) {
       cateringBox.classList.remove('hidden');
       depositCheck.checked = false;
     } else {
       cateringBox.classList.add('hidden');
       depositCheck.checked = false;
     }

     calculateOrderTotal();

     document.getElementById('orderModal').classList.remove('hidden');
     document.body.style.overflow = 'hidden';
   }

   function closeOrderModal() {
     document.getElementById('orderModal').classList.add('hidden');
     document.body.style.overflow = 'auto';
   }

   function updateQuantity(change) {
     const newQty = currentOrder.quantity + change;
     if (newQty >= 1) {
       currentOrder.quantity = newQty;
       document.getElementById('orderQtyDisplay').textContent = currentOrder.quantity;
       calculateOrderTotal();
     }
   }

   function calculateOrderTotal() {
     const hintText = document.getElementById('qtyHintText');

     if (currentOrder.unitPrice === 0) {
       document.getElementById('modalMealsSubtotal').textContent = 'حسب الاختيار';
       document.getElementById('modalDeliveryFee').textContent = '75 ج (أو مجاني لوجبتين+)';
       document.getElementById('modalItemPrice').textContent = 'يحدد مع الإدارة';
       hintText.textContent = 'توصيل مجاني عند طلب وجبتين فأكثر!';
       currentOrder.total = 0;
       return;
     }

     let addonsSum = 0;
     document.querySelectorAll('.addon-checkbox:checked').forEach(cb => {
       addonsSum += parseFloat(cb.dataset.price || 0);
     });

     const mealsSubtotal = currentOrder.unitPrice * currentOrder.quantity;
     const deliveryFee = currentOrder.quantity >= 2 ? 0 : DELIVERY_STANDARD_FEE;

     document.getElementById('modalMealsSubtotal').textContent = `${mealsSubtotal} جنيه (${currentOrder.quantity} × ${currentOrder.unitPrice})`;
     
     if (currentOrder.quantity >= 2) {
       document.getElementById('modalDeliveryFee').innerHTML = `<span class="text-emerald-400 font-black">مجاناً 🎉 (وفرت 75 ج)</span>`;
       hintText.innerHTML = `<span class="text-emerald-400 font-bold">مبروك! حصلت على توصيل مجاني 🎉</span>`;
     } else {
       document.getElementById('modalDeliveryFee').textContent = `${DELIVERY_STANDARD_FEE} جنيه`;
       hintText.textContent = `أضف وجبة أخرى واحصل على توصيل مجاني (وفر 75 ج)!`;
     }

     currentOrder.total = mealsSubtotal + addonsSum + deliveryFee;
     document.getElementById('modalItemPrice').textContent = `${currentOrder.total} جنيه`;
   }

   function toggleTimingInput(show) {
     const field = document.getElementById('laterTimeField');
     const input = document.getElementById('custCustomTime');
     if (show) {
       field.classList.remove('hidden');
       input.required = true;
       input.focus();
     } else {
       field.classList.add('hidden');
       input.required = false;
       input.value = '';
     }
   }

   function handleFormSubmit(event) {
     event.preventDefault();

     const name = document.getElementById('custName').value.trim();
     const area = document.getElementById('custArea').value;
     const address = document.getElementById('custAddress').value.trim();
     const timingType = document.querySelector('input[name="timingType"]:checked').value;
     const customTime = document.getElementById('custCustomTime').value.trim();
     const payment = document.querySelector('input[name="paymentMethod"]:checked').value;
     const depositAgreement = document.getElementById('depositAgreement');

     if (!name || !area || !address) {
       alert("يرجى ملء جميع البيانات الإلزامية.");
       return;
     }

     if (timingType === 'لاحقاً' && !customTime) {
       alert("يرجى كتابة الموعد المحدد المطلوب للاستلام.");
       return;
     }

     if (currentOrder.isCatering && !depositAgreement.checked) {
       alert("تنبيه: حجز العزومات يتطلب الموافقة على سداد العربون لتأكيد الطلب.");
       depositAgreement.focus();
       return;
     }

     const selectedAddons = [];
     document.querySelectorAll('.addon-checkbox:checked').forEach(cb => {
       selectedAddons.push(`${cb.dataset.name} (+${cb.dataset.price} ج)`);
     });

     const timingFinal = timingType === 'لاحقاً' ? `لاحقاً (${customTime})` : 'الآن (بأسرع وقت)';
     const totalText = currentOrder.total > 0 ? `${currentOrder.total} جنيه` : 'يحدد مع الإدارة';
     const deliveryText = currentOrder.quantity >= 2 ? 'مجاني 🎉 (عرض الوجبتين)' : `${DELIVERY_STANDARD_FEE} جنيه`;

     let msg = `مرحباً "ملح وملاح" 👋\n`;
     msg += `أود تأكيد طلب جديد من خلال المنيو:\n\n`;
     msg += `🍽️ *تفاصيل الطلب:*\n`;
     msg += `• الوجبة: *${currentOrder.title}*\n`;
     msg += `• الكمية: *${currentOrder.quantity} وجبة*\n`;
     msg += `• سعر الوجبات: *${currentOrder.unitPrice * currentOrder.quantity} جنيه*\n`;
     
     if (selectedAddons.length > 0) {
       msg += `• الإضافات: *${selectedAddons.join(' + ')}*\n`;
     }
     
     msg += `• رسوم التوصيل: *${deliveryText}*\n`;
     msg += `💰 *الإجمالي النهائي المطلوب:* *${totalText}*\n\n`;
     msg += `👤 *بيانات العميل والتوصيل:*\n`;
     msg += `• الاسم: *${name}*\n`;
     msg += `• منطقة التوصيل: *${area}*\n`;
     msg += `• العنوان بالتفصيل: *${address}*\n`;
     msg += `• موعد الاستلام: *${timingFinal}*\n`;
     msg += `• طريقة الدفع: *${payment}*\n`;

     if (currentOrder.isCatering) {
       msg += `\n⚠️ *ملاحظة حجز عزومة:* تم الإقرار بدفع عربون الحجز، برجاء إرسال تفاصيل التحويل لتأكيد التجهيز.`;
     }

     const whatsappUrl = `https://wa.me/${WHATSAPP_PHONE}?text=${encodeURIComponent(msg)}`;
     
     closeOrderModal();
     window.open(whatsappUrl, '_blank');
   }

   window.onclick = function(event) {
     const modal = document.getElementById('orderModal');
     if (event.target === modal) {
       closeOrderModal();
     }
   };
 </script>
